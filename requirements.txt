# Hermes TradeDesk — required Python dependencies
# -----------------------------------------------------------------------------
# This list is split into two sections:
#   [main]   - Dependencies installed by pip with normal dependency resolution.
#              These MUST be compatible with the host Hermes's constraints.
#   [lighter] - Lighter dependency installed SEPARATELY with --no-deps.
#
# Why split:
#   hermes-agent 0.19.0 requires `urllib3<3,>=2.7.0`.
#   lighter-sdk 1.1.2 declares `urllib3<2.1.0,>=1.25.3` in its METADATA.
#   These two constraints have NO overlap. If we install lighter-sdk
#   through the normal pip resolver, it downgrades urllib3 to <2.1.0
#   (e.g. 2.0.7), violating Hermes's urllib3 requirement and breaking
#   the gateway at runtime.
#
#   The TradeDesk installer therefore:
#     - Installs the [main] section with normal dependency resolution.
#     - Installs lighter-sdk==1.1.2 SEPARATELY with
#       `pip install --no-deps 'lighter-sdk==1.1.2'` so its resolver
#       never touches urllib3.
#     - Before and after the Lighter install, verifies that the host's
#       urllib3 is still in the Hermes-compatible range (<3,>=2.7.0).
# -----------------------------------------------------------------------------

# Numerical library (transitive; pinned for reproducibility)
# apexomni==3.3.1 does 'import numpy as np' at module level in
# http_private_sign.py, so numpy is REQUIRED for the [Apex] balance
# to work. Without numpy, the Apex _balance call raises
# 'ModuleNotFoundError: No module named "numpy"'.
numpy==2.4.6

# Telegram bot framework. Compatible with hermes-agent 0.19.0.
python-telegram-bot==22.6

# ===== [lighter] Special-cased install: --no-deps =====
# lighter-sdk is the only package whose METADATA declares
# urllib3<2.1.0,>=1.25.3. Installing it through the normal resolver
# would downgrade Hermes's urllib3 from 2.7.0 to 2.0.7. Instead, the
# installer installs it with --no-deps AFTER the [main] dependencies,
# with explicit urllib3 verification before and after.
# The line below is informational; the installer reads the marker
# `lighter: lighter-sdk==1.1.2` and handles it specially.
lighter: lighter-sdk==1.1.2
